// Helper: check if any quest is active between two IDs
function	script	F_ep21_queststatus_between	{
	.@start = getarg(0);
	.@end   = getarg(1);
	.@i     = .@start;
	while (.@i <= .@end) {
		if (isbegin_quest(.@i) > 0)
			return 1;
		.@i++;
	}
	return 0;
}

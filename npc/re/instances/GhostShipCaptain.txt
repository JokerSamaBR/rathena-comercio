jor_albe,195,179,4	script	Mylon#01	10529,{

	//Cooldown
	.@playtime = checkquest(16823,PLAYTIME);
	if(.@playtime == 2) erasequest 16823;
	if(.@playtime == 0 || .@playtime == 1){
		mes "[Ilze]";
		mes "This instance is still on cooldown. Please come back after the cooldown ends.";
		close;
	}

	if (getpartyleader(getcharid(1),2)!=getcharid(0)) {
		mes "Only the party leader may enter this instance alone";
		close;
	}

	set .@party_id,getcharid(1);
	set .@p_name$,getpartyname(.@party_id);

	if (!.@party_id) {
		mes "[Mylon]";
		mes "Where are your party members?";
		close;
	}
	
	.@md_name$ = "Ghost Ship Captain";
	
	if (getcharid(0) == getpartyleader(.@party_id,2))
		set @menu$, "Create "+.@md_name$+":Enter "+.@md_name$;
	else
		set @menu$, ":Enter "+.@md_name$;

	mes "[Mylon]";
	mes "Instance rewards:";
	mes "^i[102947] ^i[1001618] ^i[1001250] ^i[1001034] ^i[1001663] ^i[1001660] ^i[1001657] ^i[1001654]";
	mes "What would you like to do?";
	next;
	.@i = select(@menu$)-1;
	switch(.@i) {
	case 0:
		if(BaseLevel < 250) {
			mes "[Mylon]";
			mes "It's very dangerous inside..."; 
			mes "Come back after reaching Lv.250..."; 
			close;
		}
		
		if (instance_create(.@md_name$) < 0) {
			mes "[Mylon]";
			mes "Party name: "+.@p_name$;
			mes "Party leader: "+strcharinfo(0);
			mes "^0000ff"+.@md_name$+" ^000000- Creation failed!";
			close;
		}

		mapannounce strnpcinfo(4),"Party "+getpartyname(.@party_id)+" is preparing to challenge 【"+.@md_name$+"】instance!",bc_map,"0x33ea91";	
		mes "[Mylon]";
		mes ""+.@md_name$+" has been created~";
		mes "Please select ^0066CCEnter "+.@md_name$+"^000000.";
		
		.@iid = instance_id(IM_PARTY);
		set getinstancevar('party_id, .@iid), getcharid(1);
		//Entry time record
		set 'Instance_Annal,gettimetick(2);
		close;
	case 1:
		if(BaseLevel < 250) {
			mes "[Mylon]";
			mes "It's very dangerous inside..."; 
			mes "Come back after reaching Lv.250..."; 
			close;
		}
	if ( getinstancevar('Win,instance_id(IM_PARTY)) ) {
	mes "The instance has already started. Cannot enter again.";
	close;
	}
		switch(instance_enter(.@md_name$)) {
		case IE_OTHER:
			mes "Unknown error.";
			close;
		case IE_NOINSTANCE:
			mes .@md_name$+" instance doesn't exist.";
			mes "The leader hasn't applied for Memory Maze.";
			close;
		case IE_NOMEMBER:
			mes "Only party members who applied for "+.@md_name$+" instance may enter.";
			close;
		case IE_OK:
			//mapannounce strnpcinfo(4),"Party " + getpartyname(.@party_id) + "'s brave " + strcharinfo(0) + " has started 【"+.@md_name$+"】instance",bc_map,"0x00ff99";			
			//Quest
			if(checkquest(16826) == -1) { setquest 16826; }
			close;
		}
	}
	close;
	
OnInit:
	waitingroom "Ghost Ship Captain",0;
	end;
}

OnInstanceInit:
	//Create instance map
	'map$ = instance_mapname("1@wtgs");
	//Hide NPCs
	disablenpc instance_npcname("Ghost Ship Captain Portal#01");
	disablenpc instance_npcname("Ghost Ship Captain Portal#02");
	disablenpc instance_npcname("Ghost Ship Captain Portal#03");
	disablenpc instance_npcname("Ghost Ship Captain Portal#04");
	end;

	set 'Wtgs_Mob_01,'Wtgs_Mob_01-1;
	if ('Wtgs_Mob_01>0) end;

	instance_announce instance_id(1),"Area cleared. Please proceed to next zone",bc_all,0xD7BA98;
	end;

	set 'Wtgs_Mob_01,'Wtgs_Mob_01-1;
	if ('Wtgs_Mob_01>0) end;
	enablenpc instance_npcname("Ghost Ship Captain Portal#02");
	set 'Wtgs_Mob_01,14;
	areamonster 'map$,183,100,185,98,"--ja--",22369,3,instance_npcname("Ghost Ship Captain Event1")+"::OnCcdd3";
	areamonster 'map$,147,102,151,95,"--ja--",22369,4,instance_npcname("Ghost Ship Captain Event1")+"::OnCcdd3";
	areamonster 'map$,105,121,110,102,"--ja--",22369,7,instance_npcname("Ghost Ship Captain Event1")+"::OnCcdd3";
	instance_announce instance_id(1),"Area cleared. Please proceed to next zone",bc_all,0xD7BA98;
	end;

OnCcdd3:
	set 'Wtgs_Mob_01,'Wtgs_Mob_01-1;
	if ('Wtgs_Mob_01>0) end;
	enablenpc instance_npcname("Maristella Bart#02");
	end;

OnEnd:
	'win = 1;
	//Completion announcement time
	set 'Instance_Miao,gettimetick(2)-'Instance_Annal;
	enablenpc instance_npcname("Maristella Bart#03");
	enablenpc instance_npcname("Mylon#02");
	announce "【Ghost Ship Captain】: Party ["+strcharinfo(1)+"] led by '"+getpartyleader(getcharid(1),0)+"' completed the challenge in "+('Instance_Miao/60)+" minutes and "+('Instance_Miao%60)+" seconds",bc_all,0xD7BA98;
	end;
}

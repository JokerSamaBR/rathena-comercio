//============================================================
//= ENTRY FUNCTIONS CALLED BY GARDEN OF TIME GATES
//============================================================
function	script	F_GOT_LakeOfFire_Enter	{
	.@md_name$ = "Lake of Fire";

	if (BaseLevel < 250) {
		mes "[Dimension Prison]";
		mes "Only adventurers level 250 or higher may enter.";
		close;
	}
	if (isbegin_quest(12597) != 2) {
		mes "[Dimension Prison]";
		mes "You are not registered on the prison access list.";
		close;
	}
	if (checkweight(1201,1) == 0) {
		mes "^ff0000It seems that there are too many types and weights of items. Please organize your inventory.^000000";
		close;
	}
	if (getcharid(1) < 1) {
		mes "^0000ffPlease organize a party of at least one person and proceed.^000000";
		close;
	}

	switch( checkquest(70251,PLAYTIME) ) {
	case -1:
		break;
	case 0:
	case 1:
		mes "[Dimension Prison]";
		mes "Lake of Fire can only be entered once per day.";
		close;
	case 2:
		erasequest 70251;
		break;
	}

	mes "[Dimension Prison]";
	mes "Lake of Fire prison access.";
	next;

	.@menu$ = "";
	if (is_party_leader())
		.@menu$ = "Open Lake of Fire:";
	.@menu$ += "Enter Lake of Fire:Cancel";

	switch( select(.@menu$) ) {
	case 1:
		if (!is_party_leader()) {
			switch( instance_enter(.@md_name$) ) {
			case IE_OK:
				end;
			case IE_NOINSTANCE:
				mes "[Dimension Prison]";
				mes "The party leader has not opened the prison yet.";
				close;
			case IE_NOMEMBER:
				mes "[Dimension Prison]";
				mes "Only party members can enter.";
				close;
			case IE_OTHER:
				mes "[Dimension Prison]";
				mes "Unknown error.";
				close;
			}
		}
		if (instance_create(.@md_name$) < 0) {
			mes "[Dimension Prison]";
			mes "Failed to create Lake of Fire instance.";
			mes "Check instance_db name: Lake of Fire.";
			close;
		}
		mes "[Dimension Prison]";
		mes "Lake of Fire has been opened.";
		mes "Talk to the gate again and choose Enter.";
		close;

	case 2:
		switch( instance_enter(.@md_name$) ) {
		case IE_OK:
			setquest 70251;
			end;
		case IE_NOINSTANCE:
			mes "[Dimension Prison]";
			mes "The prison has not been opened yet.";
			close;
		case IE_NOMEMBER:
			mes "[Dimension Prison]";
			mes "Only party members can enter.";
			close;
		case IE_OTHER:
			mes "[Dimension Prison]";
			mes "Unknown error.";
			close;
		}
		end;

	case 3:
		end;
	}
	end;
}

function	script	F_GOT_HallOfLife_Enter	{
	.@md_name$ = "Hall of Life";

	if (BaseLevel < 250) {
		mes "[Dimension Prison]";
		mes "Only adventurers level 250 or higher may enter.";
		close;
	}
	if (isbegin_quest(12597) != 2) {
		mes "[Dimension Prison]";
		mes "You are not registered on the prison access list.";
		close;
	}
	if (checkweight(1201,1) == 0) {
		mes "^ff0000It seems that there are too many types and weights of items. Please organize your inventory.^000000";
		close;
	}
	if (getcharid(1) < 1) {
		mes "^0000ffPlease organize a party of at least one person and proceed.^000000";
		close;
	}
	if (countitem($GOT_HOL_KEY) < 1) {
		mes "[Dimension Prison]";
		mes "You need " + mesitemlink($GOT_HOL_KEY) + ".";
		close;
	}
	if (#GOT_HOL_LEVEL < 1)
		#GOT_HOL_LEVEL = 1;

	mes "[Dimension Prison]";
	mes "Hall of Life prison access.";
	mes "Current barrier level: ^0000ff" + #GOT_HOL_LEVEL + "^000000.";
	next;

	.@menu$ = "";
	if (is_party_leader())
		.@menu$ = "Open Hall of Life:";
	.@menu$ += "Enter Hall of Life:Cancel";

	switch( select(.@menu$) ) {
	case 1:
		if (!is_party_leader()) {
			delitem $GOT_HOL_KEY, 1;
			switch( instance_enter(.@md_name$) ) {
			case IE_OK:
				end;
			case IE_NOINSTANCE:
				getitem $GOT_HOL_KEY, 1;
				mes "[Dimension Prison]";
				mes "The party leader has not opened the prison yet.";
				close;
			case IE_NOMEMBER:
				getitem $GOT_HOL_KEY, 1;
				mes "[Dimension Prison]";
				mes "Only party members can enter.";
				close;
			case IE_OTHER:
				getitem $GOT_HOL_KEY, 1;
				mes "[Dimension Prison]";
				mes "Unknown error.";
				close;
			}
		}
		if (instance_create(.@md_name$) < 0) {
			mes "[Dimension Prison]";
			mes "Failed to create Hall of Life instance.";
			mes "Check instance_db name: Hall of Life.";
			close;
		}
		mes "[Dimension Prison]";
		mes "Hall of Life has been opened.";
		mes "Talk to the gate again and choose Enter.";
		close;

	case 2:
		delitem $GOT_HOL_KEY, 1;
		switch( instance_enter(.@md_name$) ) {
		case IE_OK:
			warp $GOT_HOL_MAP$,53,62;
			end;
		case IE_NOINSTANCE:
			getitem $GOT_HOL_KEY, 1;
			mes "[Dimension Prison]";
			mes "The prison has not been opened yet.";
			close;
		case IE_NOMEMBER:
			getitem $GOT_HOL_KEY, 1;
			mes "[Dimension Prison]";
			mes "Only party members can enter.";
			close;
		case IE_OTHER:
			getitem $GOT_HOL_KEY, 1;
			mes "[Dimension Prison]";
			mes "Unknown error.";
			close;
		}
		end;

	case 3:
		end;
	}
	end;
}

1@f_lake,0,0,0	script	GOT_LakeOfFire#Ctrl	-1,{
OnInstanceInit:
	.@map$ = instance_mapname($GOT_LOF_MAP$);

	setd "$@GOT_LOF_CLEAR_" + instance_id(), 0;
	setd "$@GOT_LOF_BOSS_COUNT_" + instance_id(), 4;

	mapannounce .@map$, "Lake of Fire: the four divine prisoners remain sealed.", bc_map;

	disablenpc instance_npcname("Lake Exit Portal#got");

	end;

OnBossDead:
	.@left = getd("$@GOT_LOF_BOSS_COUNT_" + instance_id()) - 1;
	setd "$@GOT_LOF_BOSS_COUNT_" + instance_id(), .@left;

	.@map$ = instance_mapname($GOT_LOF_MAP$);

	if (.@left > 0) {
		mapannounce .@map$, "Remaining divine prisoners: " + .@left + ".", bc_map;
		end;
	}

	setd "$@GOT_LOF_CLEAR_" + instance_id(), 1;

	mapannounce .@map$, "All divine prisoners have been suppressed. The dimensional exit has appeared.", bc_map;

	enablenpc instance_npcname("Lake Exit Portal#got");
	end;
}

//============================================================
//= HALL OF LIFE INSTANCE - BARRIER SCALING VERSION
//= Rigel is spawned dynamically when player enters instance map.
//= Battle talk timer is inside instance controller to avoid map "" error.
//============================================================

1@ba_go	mapflag	loadevent

1@ba_go,0,0,0	script	GOT_HallOfLife#Ctrl	-1,{
OnInstanceInit:
	.@map$ = instance_mapname($GOT_HOL_MAP$);

	setd "$@GOT_HOL_CLEAR_" + instance_id(), 0;
	setd "$@GOT_HOL_SPAWNED_" + instance_id(), 0;
	setd "$@GOT_HOL_BARRIER_" + instance_id(), 1;
	$@GOT_HOL_CLEAR_ALWAYS = 0;

	mapannounce .@map$, "Hall of Life: the dimensional criminal is waiting inside the prison.", bc_map;

	initnpctimer;
	end;

OnPCLoadMapEvent:
	.@map$ = instance_mapname($GOT_HOL_MAP$);

	if (strcharinfo(3) != .@map$)
		end;

	if (getd("$@GOT_HOL_SPAWNED_" + instance_id()) == 1)
		end;

	if (#GOT_HOL_LEVEL < 1)
		#GOT_HOL_LEVEL = 1;

	if (#GOT_HOL_LEVEL > 20)
		#GOT_HOL_LEVEL = 20;

	.@lv = #GOT_HOL_LEVEL;

	setd "$@GOT_HOL_BARRIER_" + instance_id(), .@lv;
	setd "$@GOT_HOL_SPAWNED_" + instance_id(), 1;

	mapannounce .@map$, "Barrier Level " + .@lv + " has been applied to Rigel.", bc_map;

	.@gid = monster(
		.@map$,
		53,93,
		"Rigel",
		22174,
		1,
		instance_npcname("GOT_HallOfLife#Ctrl")+"::OnBossDead"
	);

	.@hp_rate = 100 + ((.@lv - 1) * 40);
	.@atk_rate = 100 + ((.@lv - 1) * 15);

	getunitdata .@gid, .@data;

	.@maxhp = .@data[UDT_MAXHP] * .@hp_rate / 100;
	.@atkmin = .@data[UDT_ATKMIN] * .@atk_rate / 100;
	.@atkmax = .@data[UDT_ATKMAX] * .@atk_rate / 100;

	setunitdata .@gid, UDT_MAXHP, .@maxhp;
	setunitdata .@gid, UDT_HP, .@maxhp;
	setunitdata .@gid, UDT_ATKMIN, .@atkmin;
	setunitdata .@gid, UDT_ATKMAX, .@atkmax;

	mapannounce .@map$, "Rigel's dimensional barrier is empowering his body.", bc_map;
	end;

OnTimer15000:
	.@map$ = instance_mapname($GOT_HOL_MAP$);

	if (getd("$@GOT_HOL_CLEAR_" + instance_id()) != 1
		&& mobcount(.@map$, instance_npcname("GOT_HallOfLife#Ctrl")+"::OnBossDead") > 0) {

		switch(rand(1,6)) {
		case 1:
			mapannounce .@map$, "Rigel: Your life is brief, but its resistance is amusing.", bc_map;
			break;
		case 2:
			mapannounce .@map$, "Rigel: A dimension is nothing more than a body waiting to be worn.", bc_map;
			break;
		case 3:
			mapannounce .@map$, "Rigel: Struggle harder. I want to know how much your soul can endure.", bc_map;
			break;
		case 4:
			mapannounce .@map$, "Rigel: The Constellation of Life was weak. I merely corrected that mistake.", bc_map;
			break;
		case 5:
			mapannounce .@map$, "Rigel: Every wound you open becomes another door for me.", bc_map;
			break;
		case 6:
			mapannounce .@map$, "Rigel: Do not call this murder. I am collecting unfinished lives.", bc_map;
			break;
		}
	}

	initnpctimer;
	end;

OnBossDead:
	.@map$ = instance_mapname($GOT_HOL_MAP$);

	setd "$@GOT_HOL_CLEAR_" + instance_id(), 1;
	setd "$@GOT_HOL_SPAWNED_" + instance_id(), 0;
	$@GOT_HOL_CLEAR_ALWAYS = 1;

	.@lv = getd("$@GOT_HOL_BARRIER_" + instance_id());

	mapannounce .@map$, "Rigel: ...So this is the limit of the life I stole.", bc_map;
	mapannounce .@map$, "Hall of Life Barrier Level " + .@lv + " has been cleared.", bc_map;
	mapannounce .@map$, "Choose your result at the terminal or leave through the exit.", bc_map;
	end;
}

1@ba_go,50,52,2	script	Hall of Life Result#got	4_SYSTEM_BOX,{
	if (getd("$@GOT_HOL_CLEAR_" + instance_id()) != 1 && $@GOT_HOL_CLEAR_ALWAYS != 1) {
		mes "[Hall of Life Result]";
		mes "Suppress the dimensional criminal first.";
		close;
	}
	if (#GOT_HOL_LEVEL < 1)
		#GOT_HOL_LEVEL = 1;
	if (#GOT_HOL_LEVEL > 20)
		#GOT_HOL_LEVEL = 20;

	.@lv = #GOT_HOL_LEVEL;
	mes "[Hall of Life Result]";
	mes "Current barrier level: ^0000ff" + .@lv + "^000000.";
	next;

	.@menu$ = "Leave";
	if (.@lv < 20)
		.@menu$ += ":Increase barrier level";
	else
		.@menu$ += ":Barrier level is already max";
	.@menu$ += ":Receive weekly reward";

	switch(select(.@menu$)) {
	case 1:
		warp "t_garden",173,235;
		end;
	case 2:
		if (.@lv < 20) {
			#GOT_HOL_LEVEL++;
			mes "[Hall of Life Result]";
			mes "Barrier level increased to ^0000ff" + #GOT_HOL_LEVEL + "^000000.";
			close2;
		}
		warp "t_garden",173,235;
		end;
	case 3:
		switch( checkquest(70252,PLAYTIME) ) {
		case 2:
			erasequest 70252;
			break;
		}

		if (#GOT_HOL_REWARD_CD > gettimetick(2)) {
			mes "[Hall of Life Result]";
			mes "You have already received the weekly reward.";
			mes "Remaining seconds: ^ff0000" + (#GOT_HOL_REWARD_CD - gettimetick(2)) + "^000000.";
			close;
		}

		getitem $GOT_CIRCULATION_LIFE, 5 + .@lv;
		getitem $GOT_FRUIT_BIRTH, 2;
		getitem $GOT_FRUIT_EXTINCTION, 2;
		getitem $GOT_TRUTH_LIFE, 1;
		getitem $GOT_LIGHT_LIFE, 3 + (.@lv / 2);
		if (rand(100) < 10)
			getitem $GOT_GENESIS_LIFE, 1;
		if (rand(100) < 30)
			getitem $GOT_RUNE_STONE, 1;
		getexp 10000000 * .@lv, 5000000 * .@lv;
		#GOT_HOL_REWARD_CD = gettimetick(2) + $GOT_HOL_REWARD_DELAY;
		setquest 70252;
		mes "[Hall of Life Result]";
		mes "Weekly reward received.";
		close2;
		warp "t_garden",173,235;
		end;
	}
}

	.@mobs_page_size = .@mobs_menu_page_size;

	if(.@mobs_count == 0){
		clear;
		mes .@menu$;
		mes callfunc("F_LangText", "Sorry, there are no monsters on this map.", "Lo siento, no hay monstruos en este mapa.", "Desculpe, não h? monstros neste mapa.");
		next;
		goto(config_menu);
	}

	.@mobs_page_count = (.@mobs_total + .@mobs_page_size - 1) / .@mobs_page_size;
	.@mobs_menu_list$ = "[Cancel]:" + .@mobs_menu_list$;
	if(.@mobs_page > 0)
		.@mobs_menu_list$ += ":[Previous Page]";
	if(.@mobs_page + 1 < .@mobs_page_count)
		.@mobs_menu_list$ += ":[Next Page]";

	.@i = select(.@mobs_menu_list$);
	if(.@i == 1)
		goto(config_menu);

	.@i--;
	if(.@i <= .@mobs_count){
		if(.@mobs_menu_state[.@mobs_menu_ids[.@i]])
			aa_setautoattack("19;"+ .@mobs_menu_ids[.@i] + ";0");
		else
			aa_setautoattack("19;"+ .@mobs_menu_ids[.@i] + ";1");
		goto(monsters_menu_page);
	}

	.@navigation_index = .@mobs_count + 1;
	if(.@mobs_page > 0 && .@i == .@navigation_index){
		.@mobs_page--;
		goto(monsters_menu_page);
	}
	if(.@mobs_page + 1 < .@mobs_page_count && .@i == .@navigation_index + (.@mobs_page > 0)){
		.@mobs_page++;
		goto(monsters_menu_page);
	}

	goto(config_menu);

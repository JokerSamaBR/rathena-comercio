-	script	Adventurer_Solo_Pack_Cmd	-,{
	OnSoloPackCmd:
		// Call the Functions for the flow converse	
		callfunc("Solo_Pack_Converse");
	end;

		bindatcmd "starterpack", strnpcinfo(3) + "::OnSoloPackCmd";

function	script	Solo_Pack_Converse	{
	// Get the list of items in your inventory.
	getinventorylist;
	
	// Get the weight of the player
	.@weight = 100 * Weight / MaxWeight;

	// Get the character id of the player who is this command
	.@char_id = getcharid(3);

	if (alreadyReceived[.@char_id]) {
		mes callfunc("F_LangText", "[Starter Pack Distributor]", "[Distribuidor del Pack Inicial]", "[Distribuidor do Pack Iniciante]");
    	mes callfunc("F_LangText", "You have already received your Starter Pack items!", "¡Ya recibiste los objetos de tu Pack Inicial!", "Você já recebeu os itens do seu Pack Iniciante!");
		mes callfunc("F_LangText", "^FF0000* Remember! All items are character-bound!! *^000000", "^FF0000* ¡Recuerda! ¡Todos los objetos están vinculados al personaje! *^000000", "^FF0000* Lembre-se! Todos os itens são vinculados ao personagem! *^000000");
    	close;
	}
	
	// Check the Weight of player
	if(.@weight > 50){
		mes callfunc("F_LangText", "[Starter Pack Distributor]", "[Distribuidor del Pack Inicial]", "[Distribuidor do Pack Iniciante]");
		mes callfunc("F_LangText", "^FF0000You must have at least 50 percent free weight to receive the Starter Pack.^000000", "^FF0000Debes tener al menos 50 por ciento de peso libre para recibir el Pack Inicial.^000000", "^FF0000Você precisa ter pelo menos 50 por cento de peso livre para receber o Pack Iniciante.^000000");
		mes " ";
		mes callfunc("F_LangText", "You are currently at ", "Actualmente estás en ", "Atualmente você está com ") + .@weight + callfunc("F_LangText", " percent of your maximum weight.", " por ciento de tu peso máximo.", " por cento do seu peso máximo.");
		close;
	}

	// Check the status of the player first if they have to many items at inventory
	if ( @inventorylist_count > 40 ) {
		mes callfunc("F_LangText", "[Starter Pack Distributor]", "[Distribuidor del Pack Inicial]", "[Distribuidor do Pack Iniciante]");
		mes callfunc("F_LangText", "You have 40 different items in your inventory.", "Tienes 40 objetos diferentes en tu inventario.", "Você possui 40 itens diferentes no inventário.");
		mes callfunc("F_LangText", "Please store them in Kafra Storage first!", "¡Guárdalos primero en el almacenamiento Kafra!", "Guarde-os primeiro no Armazém Kafra!");
		close;
	}

	mes callfunc("F_LangText", "[Starter Pack Distributor]", "[Distribuidor del Pack Inicial]", "[Distribuidor do Pack Iniciante]");
    mes callfunc("F_LangText", "Hello! Would you like to receive your Starter Pack items?", "¡Hola! ¿Quieres recibir los objetos de tu Pack Inicial?", "Olá! Você deseja receber os itens do seu Pack Iniciante?");
    next;
		
	if(select(callfunc("F_LangText", "Yes", "Sí", "Sim") + ":" + callfunc("F_LangText", "No", "No", "Não")) == 2) {
		mes callfunc("F_LangText", "[Starter Pack Distributor]", "[Distribuidor del Pack Inicial]", "[Distribuidor do Pack Iniciante]");        	
		mes callfunc("F_LangText", "Alright, come back if you change your mind.", "De acuerdo, vuelve si cambias de opinión.", "Certo, volte se mudar de ideia.");
        close;
    }

    // Flag for player that he/she already received the Gpack
    // For Debug. Just put it below the .char_id and set it to 0
	set alreadyReceived[.@char_id], 1;

	// Call the Function for giving the job-specific Starter Pack item
	callfunc("Solo_Pack_Func");
}
